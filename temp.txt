        /// 导出当前模型上标记了 [BindProperty] 的属性，供前端初始化/回传使用。
        /// </summary>
        public object Export()
        {
            IDictionary<string, object> data = new ExpandoObject();
            var properties = GetType().GetProperties(BindingFlags.Instance | BindingFlags.Public);

            foreach (var property in properties)
            {
                if (!property.CanRead || property.GetIndexParameters().Length > 0)
                    continue;
                if (property.GetCustomAttribute<BindPropertyAttribute>() == null)
                    continue;

                data[property.Name] = property.GetValue(this);
            }

            return data;
        }

        [HttpApi("Point","获取检查对象点位数据", AuthLogin = false)]
        public static APIResult GetCheckObjectPoints(string name = null, long? orgId = null, bool? isDel = false, string region = null, int maxCount = 500)
        {
            if (maxCount <= 0)     maxCount = 100;
            if (maxCount > 5000)   maxCount = 5000;

            // 获取点位数据
            var q = CheckObject.Set.AsNoTracking()  .Where(t => !string.IsNullOrWhiteSpace(t.Gps));
            if (isDel != null)           q = q.Where(t => t.IsDel == isDel.Value);
            if (orgId.IsNotEmpty())      q = q.Where(t => t.OrgId == orgId.Value);
            if (name.IsNotEmpty())       q = q.Where(t => (t.Name ?? string.Empty).Contains(name.Trim()));
            var items = q
                .OrderBy(t => t.Id)
                .Take(maxCount * 4)
                .Select(t => new
                {
                    t.Id,
                    t.Name,
                    t.Address,
                    t.Gps,
                    t.SocialCreditId,
                    t.RiskLevel,
                    t.ObjectType,
                    t.Scale,
                    t.OrgId,
                    t.DutyMan,
                })
                .ToList();

            // 过滤掉不合法的坐标和不在指定区域内的点位。
            var regioner = GisRegion.Parse(region);
            var list = new List<object>();
            foreach (var item in items)
            {
                if (!GisHelper.TryParseLngLat(item.Gps, out var lng, out var lat))
                    continue;
                if (regioner != null && !regioner.Contains(lng, lat))
                    continue;

                list.Add(new
                {
                    id = item.Id,
                    name = item.Name,
                    gps = item.Gps,
                    addr = item.Address,
                    lng,
                    lat,
                    dataJson = new
                    {
                        item.SocialCreditId,
                        item.RiskLevel,
                        item.ObjectType,
                        item.Scale,
                        item.OrgId,
                        item.DutyMan,
                    }
                });

                if (list.Count >= maxCount)
                    break;
            }

            return list.ToResult();
        }

@section script {
    <script>
        (function () {
            const storageKey = '__gis_geometry_editor_reference_gps';
            const storages = [window.sessionStorage, window.localStorage].filter(Boolean);
            const gpsSelectors = [
                '[data-ele-control-id="field:gps"]',
                '.gps-picker'
            ];
            const geoJsonSelectors = [
                '[data-ele-control-id="field:geoData"]',
                '.geojson-picker'
            ];

            function findHost(selectors) {
                for (let i = 0; i < selectors.length; i += 1) {
                    const el = document.querySelector(selectors[i]);
                    if (el) return el;
                }
                return null;
            }

            function getGpsInputValue() {
                const host = findHost(gpsSelectors);
                if (!host) return '';
                const input = host.querySelector('input,textarea');
                return String(input?.value || '').trim();
            }

            function syncReferenceGps() {
                try {
                    const gps = getGpsInputValue();
                    storages.forEach(storage => {
                        if (gps) storage.setItem(storageKey, gps);
                        else storage.removeItem(storageKey);
                    });
                } catch {
                }
            }

            function bindGpsSync() {
                const host = findHost(gpsSelectors);
                if (!host) return false;
                const input = host.querySelector('input,textarea');
                if (input && !input.__gisGpsSyncBound) {
                    input.__gisGpsSyncBound = true;
                    ['input', 'change', 'blur'].forEach(name => {
                        input.addEventListener(name, syncReferenceGps, true);
                    });
                }
                syncReferenceGps();
                return !!input;
            }

            ['pointerdown', 'mousedown', 'click'].forEach(name => {
                document.addEventListener(name, function (evt) {
                    if (!evt.target || !evt.target.closest) return;
                    if (evt.target.closest(geoJsonSelectors.join(','))) {
                        syncReferenceGps();
                    }
                }, true);
            });

            let retry = 0;
            const timer = setInterval(() => {
                retry++;
                const ok = bindGpsSync();
                if (ok || retry >= 20) {
                    clearInterval(timer);
                }
            }, 300);
        })();
    </script>
}

        /// <summary>把路径拼接为安全绝对路径</summary>
        public static string Combine(string root, params string[] parts)
        {
            if (parts == null || parts.Length == 0) return root ?? "";
            var all = new string[parts.Length + 1];
            all[0] = root;
            Array.Copy(parts, 0, all, 1, parts.Length);
            return GetFullPath(Path.Combine(all.Where(s => !string.IsNullOrEmpty(s)).ToArray()));
        }
