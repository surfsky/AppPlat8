        private IActionResult MoveMenu(ElePayload payload, int direction)
        {
            if (payload?.SelectedCount != 1 || payload.FirstSelectedId is not long id || id <= 0)
                return BuildResult(400, "请只选择一个菜单");

            var selected = Menu.Get(id);
            if (selected == null)
                return BuildResult(404, "菜单不存在");

            var siblings = Menu.Set
                .Where(menu => menu.ParentId == selected.ParentId)
                .OrderBy(menu => menu.SortId)
                .ThenBy(menu => menu.Id)
                .ToList();
            var index = siblings.FindIndex(menu => menu.Id == selected.Id);
            var targetIndex = index + direction;
            if (index < 0 || targetIndex < 0 || targetIndex >= siblings.Count)
                return BuildResult(400, direction < 0 ? "已经是最上面的菜单" : "已经是最下面的菜单");

            siblings.RemoveAt(index);
            siblings.Insert(targetIndex, selected);
            var sortChanges = new List<object>();
            for (var i = 0; i < siblings.Count; i++)
            {
                var sibling = siblings[i];
                var newSortId = (i + 1) * 10;
                if (sibling.SortId == newSortId)
                    continue;

                sibling.SortId = newSortId;
                sibling.Save();
                sortChanges.Add(new { id = sibling.Id, sortId = newSortId });
            }

            Menu.ClearCache();
            return BuildResult(0, direction < 0 ? "上移成功" : "下移成功", new
            {
                reload = true,
                id = selected.Id,
                sortChanges
            });
        }
